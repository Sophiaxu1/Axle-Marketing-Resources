3. Create a new file client/src/hooks/useImageLibrary.ts with this content:
import { useEffect, useState } from "react";
import { supabase } from "@/lib/supabase";

export interface LibraryImage {
  name: string;
  url: string;
}

export function useImageLibrary(brand: string) {
  const [images, setImages] = useState<LibraryImage[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchImages() {
      setLoading(true);
      const folderMap: Record<string, string> = {
        axlerator: "Axlerator/Image Library",
        axle: "Axle/Image Library",
        art: "Axle Research and Technologies/Image Library",
      };
      const folder = folderMap[brand];
      if (!folder) { setLoading(false); return; }

      const { data, error } = await supabase.storage
        .from("Marketing Assets")
        .list(folder, { limit: 200 });

      if (error || !data) { setLoading(false); return; }

      const imageList = data
        .filter((f) => f.name !== ".emptyFolderPlaceholder")
        .map((f) => ({
          name: f.name,
          url: `https://yuiqlcfnknrjsbywmsas.supabase.co/storage/v1/object/public/Marketing%20Assets/${encodeURIComponent(folder)}/${encodeURIComponent(f.name)}`,
        }));

      setImages(imageList);
      setLoading(false);
    }
    fetchImages();
  }, [brand]);

  return { images, loading };
}
