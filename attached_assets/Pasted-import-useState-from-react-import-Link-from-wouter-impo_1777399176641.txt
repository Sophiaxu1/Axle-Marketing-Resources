function LibraryFolder({ folder, parentSlug }: { folder: { label: string; assets: LibraryAsset[] }; parentSlug: string }) {
  const [expanded, setExpanded] = useState(false);
  const folderSlug = folder.label.toLowerCase().replace(/\s+/g, "-");
  const previewCount = 4;
  const hasMore = folder.assets.length > previewCount;
  const visibleAssets = expanded ? folder.assets : folder.assets.slice(0, previewCount);

  if (folder.assets.length === 0) {
    return (
      <div className="px-5 py-4">
        <p className="text-sm font-medium mb-3" data-testid={`text-${parentSlug}-${folderSlug}`}>{folder.label}</p>
        <div className="rounded border border-dashed border-border p-6 text-center">
          <p className="text-sm text-muted-foreground">No images yet</p>
        </div>
      </div>
    );
  }

  return (
    <div className="px-5 py-4">
      <div className="flex items-center justify-between mb-3">
        <p className="text-sm font-medium" data-testid={`text-${parentSlug}-${folderSlug}`}>
          {folder.label} ({folder.assets.length})
        </p>
      </div>
      <div className="grid grid-cols-2 gap-2">
        {visibleAssets.map((asset) => (
          <div
            key={asset.name}
            className="group relative rounded border border-border overflow-hidden bg-muted/30"
            data-testid={`asset-${parentSlug}-${asset.name}`}
          >
            {asset.type === "image" ? (
              <>
                <img
                  src={asset.file}
                  alt={asset.name}
                  className="w-full h-20 object-cover"
                  loading="lazy"
                />
                <a
                  href={`/api/download?file=${encodeURIComponent(asset.file)}`}
                  className="absolute inset-0 flex items-center justify-center bg-black/0 group-hover:bg-black/40 transition-colors"
                  data-testid={`download-${parentSlug}-${asset.name}`}
                >
                  <Download className="w-4 h-4 text-white opacity-0 group-hover:opacity-100 transition-opacity" />
                </a>
              </>
            ) : (
              <div className="relative">
                <video
                  src={asset.file}
                  controls
                  preload="metadata"
                  className="w-full h-20 object-cover bg-black"
                />
                <a
                  href={`/api/download?file=${encodeURIComponent(asset.file)}`}
                  className="absolute top-1 right-1 p-1 rounded bg-black/50 hover:bg-black/70 transition-colors"
                  data-testid={`download-${parentSlug}-${asset.name}`}
                >
                  <Download className="w-3 h-3 text-white" />
                </a>
              </div>
            )}
          </div>
        ))}
      </div>
      {hasMore && (
        <Button
          variant="ghost"
          size="sm"
          className="w-full mt-2 text-xs"
          onClick={() => setExpanded(!expanded)}
          data-testid={`btn-toggle-${parentSlug}-${folderSlug}`}
        >
          {expanded ? (
            <>Show Less <ChevronUp className="w-3 h-3 ml-1" /></>
          ) : (
            <>Show All ({folder.assets.length}) <ChevronDown className="w-3 h-3 ml-1" /></>
          )}
        </Button>
      )}
    </div>
  );
}
